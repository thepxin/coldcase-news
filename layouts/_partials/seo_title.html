{{- /* The shown in browser tabs and search results, at most 65 characters (Bing flags longer ones, Google cuts them at about 60). The headline on the page, social cards and structured data keep the full title. 1. A two-sentence house-voice headline keeps its first sentence when that is a full thought of 35+ characters ("Rockstar's GTA 6 is due November 19, 2026"). Dots in "Nov. 4" or "p.m." are not sentence ends. 2. " | ColdCaseTech" is added only when it still fits. 3. Anything longer is cut at a word break, with an ellipsis. */ -}} {{- $max := 65 -}} {{- $out := site.Title -}} {{- if and (not .IsHome) .Title -}} {{- $t := .Title | plainify | htmlUnescape | strings.TrimSpace -}} {{- with findRESubmatch `^(.{35,}?\b[A-Za-z0-9%)]{2,})\. [A-Z]` $t 1 -}} {{- $t = index (index . 0) 1 -}} {{- end -}} {{- $full := printf "%s | %s" $t site.Title -}} {{- if le (strings.RuneCount $full) $max -}} {{- $out = $full -}} {{- else if le (strings.RuneCount $t) $max -}} {{- $out = $t -}} {{- else -}} {{- $cut := "" -}} {{- $full := false -}} {{- range split $t " " -}} {{- $next := cond (eq $cut "") . (printf "%s %s" $cut .) -}} {{- if and (not $full) (le (strings.RuneCount $next) (sub $max 1)) }}{{ $cut = $next }}{{ else }}{{ $full = true }}{{ end -}} {{- end -}} {{- $out = printf "%s…" (strings.TrimRight ",;:-" $cut) -}} {{- end -}} {{- end -}} {{- $out -}}
